              Python requests Example (Total Document Verification):
            
            

              import requests


              url = "http://127.0.0.1:8000/api/v1/verify"

              headers = {"X-API-Key": "vrd_live_master_kiosk_key"}

              files = {"file": open("document_scan.jpg", "rb")}

              data = {

                "checkpoint_id": "DESKTOP-TERMINAL",

                "officer_id": "OFFICER-01",

                "force_deep_ai": "false"

              }


              resp = requests.post(url, headers=headers, files=files, data=data)

              result = resp.json()

              print(f"Verdict: {result['overall_verdict']}")

              print(f"Risk Score: {result['risk']['risk_score']}/100")

              print(f"Extracted: {result['extracted_fields']['name']}, UID: {result['extracted_fields']['document_number']}")
